cmake_minimum_required(VERSION 3.16)
project(TWalkEnumIndex LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(index_worker common/src/worker.cpp)
target_include_directories(index_worker PRIVATE common/include)
target_link_libraries(index_worker PRIVATE Threads::Threads)
target_compile_options(index_worker PRIVATE -O3 -DNDEBUG -Wall -Wextra -Wpedantic)
set_target_properties(index_worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(graph_check ../../tools/graph_check.cpp)
target_include_directories(graph_check PRIVATE common/include)
target_link_libraries(graph_check PRIVATE Threads::Threads)
target_compile_options(graph_check PRIVATE -O3 -DNDEBUG)
set_target_properties(graph_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
