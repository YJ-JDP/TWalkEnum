cmake_minimum_required(VERSION 3.16)
project(TWalkEnumGap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)
add_executable(gap_worker apps/worker.cpp)
target_include_directories(gap_worker PRIVATE common/include)
target_link_libraries(gap_worker PRIVATE Threads::Threads)
target_compile_options(gap_worker PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3;-DNDEBUG;-Wall;-Wextra>)
set_target_properties(gap_worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
